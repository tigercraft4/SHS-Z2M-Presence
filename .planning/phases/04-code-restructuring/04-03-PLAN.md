   Replace the 30 copy-paste zone switch cases (~L1790-2015) with computed offset logic:

   ```c
   /* Zone attribute range: zone 1 = 0x0021-0x0027, zone N = 0x0020 + N*0x10 to +0x06 */
   static esp_err_t handle_zone_attr(uint16_t attr_id, uint8_t v8, int16_t v16s) {
       int zone_idx = -1;
       int field_offset = -1;

       if (attr_id >= SHS_ATTR_ZONE1_ENABLED && attr_id <= SHS_ATTR_ZONE1_TYPE_CFG) {
           zone_idx = 0;
           field_offset = attr_id - SHS_ATTR_ZONE1_ENABLED;
       } else {
           for (int z = 1; z < SHS_NUM_ZONES; z++) {
               uint16_t base = 0x0020 + z * 0x10;  /* zone2=0x0030, zone3=0x0040, etc. */
               if (attr_id >= base && attr_id <= base + 6) {
                   zone_idx = z;
                   field_offset = attr_id - base;
                   break;
               }
           }
       }

       if (zone_idx < 0) return ESP_ERR_NOT_FOUND;

       shs_zone_t *zone = &s_state->zones[zone_idx];
       if (xSemaphoreTake(s_state->zone_config_mutex, pdMS_TO_TICKS(50)) != pdTRUE)
           return ESP_OK;

       switch (field_offset) {
           case 0: zone->enabled = (v8 != 0); break;       /* ENABLED offset */
           case 1: zone->x1 = v16s; break;                  /* X1 offset */
           case 2: zone->y1 = v16s; break;                  /* Y1 offset */
           case 3: zone->x2 = v16s; break;                  /* X2 offset */
           case 4: zone->y2 = v16s; break;                  /* Y2 offset */
           /* case 5: targets — read-only, skip */
           case 6: zone->type = v8; break;                   /* TYPE offset */
           default: break;
       }
       xSemaphoreGive(s_state->zone_config_mutex);

       ESP_LOGI(TAG, "Zone %d field %d updated", zone_idx + 1, field_offset);
       shs_zone_cfg_schedule_apply();
       return ESP_OK;
   }
   ```

   This replaces ~220 lines of switch cases with ~30 lines. The attribute handler's config cluster section calls `handle_zone_attr()` for any attr_id >= 0x0020 that isn't in the non-zone config range.

```c
/*
 * SHS01 Presence Sensor — Application Entry Point
 */

#include <string.h>
#include "esp_log.h"
#include "nvs_flash.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"
#include "esp_task_wdt.h"

#include "shs01.h"
#include "shs_state.h"
#include "shs_config.h"
#include "shs_zigbee.h"
#include "shs_sensor.h"
#include "shs_button.h"
#include "light_driver.h"

static const char *TAG = "SHS01";

/* Application state — single allocation, shared by all modules */
static shs_state_t g_state;

void app_main(void) {
    ESP_LOGI(TAG, "SHS01 Presence Sensor starting...");

    /* Initialize NVS */
    esp_err_t ret = nvs_flash_init();
    if (ret == ESP_ERR_NVS_NO_FREE_PAGES || ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {
        ESP_ERROR_CHECK(nvs_flash_erase());
        ret = nvs_flash_init();
    }
    ESP_ERROR_CHECK(ret);

    /* Initialize shared state */
    shs_state_init(&g_state);

    /* Initialize modules (order matters — stores state pointer) */
    shs_config_init(&g_state);
    shs_zigbee_init(&g_state);
    shs_sensor_init(&g_state);
    shs_button_init(&g_state);

    /* Load configuration from NVS */
    shs_cfg_load_from_nvs();
    shs_zone_cfg_load_from_nvs();

    /* Initialize LED */
    light_driver_init(false);

    /* Initialize task watchdog (Phase 3: 10s timeout, panic on trigger) */
    esp_task_wdt_config_t twdt_config = {
        .timeout_ms = 10000,
        .trigger_panic = true,
        .idle_core_mask = 0,
    };
    ESP_ERROR_CHECK(esp_task_wdt_init(&twdt_config));

    /* Create tasks */
    xTaskCreate(shs_zigbee_task, "zigbee", 4096, NULL, 5, NULL);
    xTaskCreate(shs_ld2410_task, "ld2410", 3072, NULL, 4, NULL);
    xTaskCreate(shs_ld2450_task, "ld2450", 4096, NULL, 4, NULL);
    xTaskCreate(shs_boot_button_task, "button", 2048, NULL, 3, NULL);
    xTaskCreate(shs_save_worker, "save", 3072, NULL, 3, NULL);

    ESP_LOGI(TAG, "All tasks started");
}
```
